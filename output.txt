extends Node

# ============================================================
# LOOP CANDIDATE DIAGNOSTIC
# ============================================================

const SAMPLE_RATE: float = 30.0
const EDGE_WINDOW: float = 0.05

const VELOCITY_DIRECTION_LIMIT: float = 0.35
const VELOCITY_MAGNITUDE_LIMIT: float = 0.35
const ROTATION_SEAM_LIMIT_DEGREES: float = 15.0

const ANIMATION_FILES: Array[String] = [
	"res://Assets/Animations/Run.glb",
	"res://Assets/Animations/Walk.glb",
	"res://Assets/Animations/Sprint.glb",
	"res://Assets/Animations/Crouch.glb",
	"res://Assets/Animations/Idle.glb"
]


func _ready() -> void:
	run_diagnostic()


# ============================================================
# MAIN
# ============================================================

func run_diagnostic() -> void:
	print("\n")
	print("============================================================")
	print("             ANIMATION LOOP CANDIDATE DIAGNOSTIC")
	print("============================================================")

	var total_animations: int = 0
	var total_candidates: int = 0

	for source_file in ANIMATION_FILES:
		print("\n")
		print("============================================================")
		print("SOURCE: ", source_file)
		print("============================================================")

		var packed_scene: PackedScene = load(source_file) as PackedScene

		if packed_scene == null:
			push_error("Could not load: " + source_file)
			continue

		var instance: Node = packed_scene.instantiate()
		add_child(instance)

		var animation_player: AnimationPlayer = find_animation_player(instance)

		if animation_player == null:
			push_error("No AnimationPlayer found: " + source_file)
			instance.queue_free()
			continue

		var file_candidates: Array[Dictionary] = []
		var file_animations: int = 0

		for animation_name in animation_player.get_animation_list():
			var animation: Animation = animation_player.get_animation(animation_name)

			if animation == null:
				continue

			var position_track: int = find_root_track(
				animation,
				Animation.TYPE_POSITION_3D
			)

			if position_track == -1:
				continue

			if animation.length <= EDGE_WINDOW * 2.0:
				continue

			if animation.track_get_key_count(position_track) < 2:
				continue

			var rotation_track: int = find_root_track(
				animation,
				Animation.TYPE_ROTATION_3D
			)

			var result: Dictionary = analyze_animation(
				animation,
				position_track,
				rotation_track
			)

			result["name"] = str(animation_name)
			result["source"] = source_file.get_file()
			result["has_loop_tag"] = has_loop_tag(str(animation_name))

			file_animations += 1
			total_animations += 1

			if is_candidate(result):
				file_candidates.append(result)

		file_candidates.sort_custom(
			func(a: Dictionary, b: Dictionary) -> bool:
				return float(a["score"]) < float(b["score"])
		)

		print("\nAnimations inspected: ", file_animations)
		print("Potential loop candidates: ", file_candidates.size())

		for candidate in file_candidates:
			total_candidates += 1
			print_candidate(candidate)

		instance.queue_free()

	print("\n")
	print("============================================================")
	print("                 DIAGNOSTIC SUMMARY")
	print("============================================================")
	print("Animations inspected: ", total_animations)
	print("Potential loop candidates: ", total_candidates)
	print("============================================================\n")


# ============================================================
# ANALYZE ONE ANIMATION
# ============================================================

func analyze_animation(
	animation: Animation,
	position_track: int,
	rotation_track: int
) -> Dictionary:

	var duration: float = animation.length

	var initial_position: Vector3 = (
		animation.position_track_interpolate(position_track, 0.0)
	)

	var final_position: Vector3 = (
		animation.position_track_interpolate(position_track, duration)
	)

	var displacement: Vector3 = final_position - initial_position

	var horizontal_displacement: Vector3 = Vector3(
		displacement.x,
		0.0,
		displacement.z
	)

	var average_velocity: Vector3 = displacement / duration

	var initial_velocity: Vector3 = estimate_velocity(
		animation,
		position_track,
		EDGE_WINDOW,
		true
	)

	var final_velocity: Vector3 = estimate_velocity(
		animation,
		position_track,
		EDGE_WINDOW,
		false
	)

	var initial_horizontal: Vector3 = Vector3(
		initial_velocity.x,
		0.0,
		initial_velocity.z
	)

	var final_horizontal: Vector3 = Vector3(
		final_velocity.x,
		0.0,
		final_velocity.z
	)

	var initial_speed: float = initial_horizontal.length()
	var final_speed: float = final_horizontal.length()

	var velocity_magnitude_difference: float = absf(
		initial_speed - final_speed
	)

	var velocity_direction_difference: float = 1.0

	if initial_speed > 0.05 and final_speed > 0.05:
		velocity_direction_difference = (
			1.0 - initial_horizontal.normalized().dot(
				final_horizontal.normalized()
			)
		) * 0.5
	else:
		velocity_direction_difference = 0.0 if (
			initial_speed <= 0.05 and final_speed <= 0.05
		) else 1.0

	var rotation_difference_degrees: float = 0.0

	if rotation_track != -1:
		var initial_rotation: Quaternion = (
			animation.rotation_track_interpolate(
				rotation_track,
				0.0
			)
		)

		var final_rotation: Quaternion = (
			animation.rotation_track_interpolate(
				rotation_track,
				duration
			)
		)

		var relative_rotation: Quaternion = (
			initial_rotation.inverse() * final_rotation
		).normalized()

		rotation_difference_degrees = rad_to_deg(
			relative_rotation.get_angle()
		)

	var score: float = (
		velocity_direction_difference * 100.0
		+ velocity_magnitude_difference * 10.0
		+ rotation_difference_degrees
	)

	return {
		"duration": duration,
		"displacement": displacement,
		"horizontal_displacement": horizontal_displacement,
		"average_velocity": average_velocity,
		"initial_velocity": initial_velocity,
		"final_velocity": final_velocity,
		"initial_speed": initial_speed,
		"final_speed": final_speed,
		"velocity_direction_difference": velocity_direction_difference,
		"velocity_magnitude_difference": velocity_magnitude_difference,
		"rotation_difference_degrees": rotation_difference_degrees,
		"score": score
	}


# ============================================================
# ROOT VELOCITY
# ============================================================

func estimate_velocity(
	animation: Animation,
	position_track: int,
	window: float,
	at_start: bool
) -> Vector3:

	var first_time: float
	var second_time: float

	if at_start:
		first_time = 0.0
		second_time = window
	else:
		first_time = animation.length - window
		second_time = animation.length

	var first_position: Vector3 = (
		animation.position_track_interpolate(
			position_track,
			first_time
		)
	)

	var second_position: Vector3 = (
		animation.position_track_interpolate(
			position_track,
			second_time
		)
	)

	return (second_position - first_position) / window


# ============================================================
# CANDIDATE FILTER
# ============================================================

func is_candidate(result: Dictionary) -> bool:

	var initial_speed: float = float(result["initial_speed"])
	var final_speed: float = float(result["final_speed"])

	var direction_difference: float = float(
		result["velocity_direction_difference"]
	)

	var magnitude_difference: float = float(
		result["velocity_magnitude_difference"]
	)

	var rotation_difference: float = float(
		result["rotation_difference_degrees"]
	)

	var average_velocity: Vector3 = result["average_velocity"]

	var average_horizontal_speed: float = Vector2(
		average_velocity.x,
		average_velocity.z
	).length()

	# Exclude clips that are clearly not locomotion speed.
	if average_horizontal_speed > 8.0:
		return false

	# Exclude large endpoint velocity discontinuities.
	if direction_difference > VELOCITY_DIRECTION_LIMIT:
		return false

	if magnitude_difference > VELOCITY_MAGNITUDE_LIMIT:
		return false

	# Exclude large accumulated root rotations.
	if rotation_difference > ROTATION_SEAM_LIMIT_DEGREES:
		return false

	# Require some actual movement, unless it is an in-place loop.
	var displacement: Vector3 = result["horizontal_displacement"]

	if displacement.length() < 0.01:
		if maxf(initial_speed, final_speed) > 0.5:
			return true

		return false

	# Moving clips must have meaningful movement.
	if average_horizontal_speed < 0.1:
		return false

	return true


# ============================================================
# PRINT
# ============================================================

func print_candidate(result: Dictionary) -> void:

	print("\n----------------------------------------")
	print("CANDIDATE: ", result["name"])

	print("Already has loop tag: ", result["has_loop_tag"])
	print("Duration: ", snappedf(float(result["duration"]), 0.001))

	print("Displacement: ", result["displacement"])
	print("Average velocity: ", result["average_velocity"])

	print("Initial velocity: ", result["initial_velocity"])
	print("Final velocity: ", result["final_velocity"])

	print(
		"Initial speed: ",
		snappedf(float(result["initial_speed"]), 0.001)
	)

	print(
		"Final speed: ",
		snappedf(float(result["final_speed"]), 0.001)
	)

	print(
		"Velocity direction difference: ",
		snappedf(float(result["velocity_direction_difference"]), 0.001)
	)

	print(
		"Velocity magnitude difference: ",
		snappedf(float(result["velocity_magnitude_difference"]), 0.001)
	)

	print(
		"Accumulated root rotation: ",
		snappedf(float(result["rotation_difference_degrees"]), 0.01),
		" degrees"
	)

	print("Diagnostic score: ", snappedf(float(result["score"]), 0.001))


# ============================================================
# HELPERS
# ============================================================

func find_animation_player(node: Node) -> AnimationPlayer:

	if node is AnimationPlayer:
		return node as AnimationPlayer

	for child in node.get_children():
		var result: AnimationPlayer = find_animation_player(child)

		if result != null:
			return result

	return null


func find_root_track(
	animation: Animation,
	track_type: Animation.TrackType
) -> int:

	for track_index in range(animation.get_track_count()):

		if animation.track_get_type(track_index) != track_type:
			continue

		var track_path: String = str(
			animation.track_get_path(track_index)
		)

		if track_path.ends_with(":root"):
			return track_index

	return -1


func has_loop_tag(animation_name: String) -> bool:

	var tokens: PackedStringArray = (
		animation_name.to_lower()
		.replace("_", " ")
		.split(" ", false)
	)

	return "loop" in tokens
